cmake_minimum_required(VERSION 3.20)
project(LumenCode VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
if(MSVC)
    # Source literals and generated grammars must not depend on the host codepage.
    add_compile_options(/utf-8 /bigobj)
endif()
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(GNUInstallDirs)
option(LUMENCODE_BUILD_QT_APP "Build the Qt/Kirigami interface" ON)
option(LUMENCODE_BUILD_CLI "Build the public Qt-backed CLI" ON)
option(LUMENCODE_BUILD_ENGINE_HELPER "Build the JSON analysis helper" ON)
option(LUMENCODE_CLI_ONLY "Compatibility: build the public CLI without the Qt GUI" OFF)
option(LUMENCODE_BUILD_WEB "Build the browser WebAssembly engine prototype" OFF)
if(LUMENCODE_BUILD_WEB)
    if(LUMENCODE_CLI_ONLY)
        message(FATAL_ERROR "Select either LUMENCODE_BUILD_WEB or LUMENCODE_CLI_ONLY")
    endif()
    if(NOT EMSCRIPTEN)
        message(FATAL_ERROR "LUMENCODE_BUILD_WEB requires the Emscripten toolchain")
    endif()
    set(LUMENCODE_BUILD_QT_APP OFF)
    set(LUMENCODE_BUILD_CLI OFF)
    set(LUMENCODE_BUILD_ENGINE_HELPER OFF)
endif()
if(LUMENCODE_CLI_ONLY)
    set(LUMENCODE_BUILD_QT_APP OFF)
    set(LUMENCODE_BUILD_CLI ON)
endif()
if(LUMENCODE_BUILD_QT_APP)
    find_package(ECM 5.80 REQUIRED NO_MODULE)
    list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
    include(KDEInstallDirs)
    include(KDECMakeSettings)
    include(KDECompilerSettings NO_POLICY_SCOPE)
    find_package(Qt5 5.15 REQUIRED COMPONENTS Quick QuickControls2 Widgets)
    find_package(KF5 REQUIRED COMPONENTS Kirigami2 CoreAddons)
elseif(LUMENCODE_BUILD_CLI)
    find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets)
endif()
add_subdirectory(engine)
if(LUMENCODE_BUILD_WEB)
    add_subdirectory(apps/web)
endif()
if(LUMENCODE_BUILD_QT_APP OR LUMENCODE_BUILD_CLI)
    add_subdirectory(apps/qt)
endif()
